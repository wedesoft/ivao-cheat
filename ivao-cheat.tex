\begin{itemize}
\item \textbf{IFR Clearance (RNAV SID)}: \\
\textit{Pilot request:} Clearance, \callsign, IFR to \destination~airport, ready to copy. \\
\textit{ATC:} \callsign, cleared to \destination~airport, \SID~departure, then as filed. Climb via the sid, expect (cruise level) 10 minutes after departure. Departure frequency (freq), squawk \squawk. \\
\textit{Pilot readback:} Cleared to \destination~airport via \SID~departure, then as filed. Climb via the SID, expect (cruise level) 10 minutes after departure, departure frequency (freq), squawk \squawk, \callsign.

\item \textbf{IFR Clearance (Vectors)}: \\
\textit{Pilot request:} Clearance, \callsign, IFR to \destination~airport, ready to copy. \\
\textit{ATC:} \callsign, cleared to \destination~airport, \SID~departure, radar vectors (fix), then as filed. Maintain (altitude), expect (cruise level) 10 minutes after departure. Departure frequency (freq), squawk \squawk. \\
\textit{Pilot readback:} Cleared to \destination~airport via \SID~departure, radar vectors (fix), then as filed. Maintain (altitude), expect (cruise level) 10 minutes after departure, departure frequency (freq), squawk \squawk, \callsign.

\item \textbf{Pushback (non-movement area)}: \\
\textit{Pilot request:} Ground, \callsign, stand/gate (gate), request push and start. \\
\textit{ATC:} Readback correct, start and push at your discretion, expect runway (rwy) for departure, advise when ready to taxi. \\
\textit{Pilot response:} Start and push at our discretion, expect runway (rwy), will advise ready to taxi, \callsign.

\item \textbf{Pushback (movement area)}: \\
\textit{Pilot request:} Ground, \callsign, stand/gate (gate), request push and start. \\
\textit{ATC:} Readback correct, start and push approved, face (direction), expect runway (rwy) for departure, advise when ready to taxi. \\
\textit{Pilot response:} Start and push approved, face (direction), expect runway (rwy), will advise ready to taxi, \callsign.

\item \textbf{Taxi}: \\
\textit{Pilot request:} Ground, \callsign, ready to taxi. \\
\textit{ATC:} \callsign, runway (rwy), taxi via (twy’s), cross/hold short runway (rwy). \\
\textit{Pilot readback:} Runway (rwy), taxi via (twy’s), cross/hold short runway (rwy), \callsign.

\item \textbf{Line-up}: \\
\textit{Pilot request:} Tower, \callsign, ready for departure. \\
\textit{ATC:} \callsign, runway (rwy), line up and wait. \\
\textit{Pilot readback:} Runway (rwy), line up and wait, \callsign.

\item \textbf{Line-up at intersection}: \\
\textit{Pilot request:} Tower, \callsign, ready for departure at (twy). \\
\textit{ATC:} \callsign, runway (rwy) at (twy), line up and wait. \\
\textit{Pilot readback:} Runway (rwy) at (twy), line up and wait, \callsign.

\item \textbf{Takeoff (RNAV)}: \\
\textit{ATC:} \callsign, RNAV (initial fix), wind (degrees) at (speed). Runway (rwy), cleared for takeoff. \\
\textit{Pilot readback:} RNAV (initial fix), runway (rwy), cleared for takeoff, \callsign.

\item \textbf{Takeoff (Vectors)}: \\
\textit{ATC:} \callsign, fly heading (hdg), wind (degrees) at (speed). Runway (rwy), cleared for takeoff. \\
\textit{Pilot readback:} Fly heading (hdg), runway (rwy), cleared for takeoff, \callsign.

\item \textbf{Takeoff at intersection}: \\
\textit{ATC:} \callsign, fly heading (hdg), wind (degrees) at (speed). Runway (rwy) at (twy), cleared for takeoff. \\
\textit{Pilot readback:} Fly heading (hdg), runway (rwy) at (twy), cleared for takeoff, \callsign.

\item \textbf{Landing}: \\
\textit{ATC:} \callsign, wind (degrees) at (speed), runway (rwy), cleared to land. \\
\textit{Pilot readback:} Runway (rwy), cleared to land, \callsign.

\item \textbf{Landing w/ sequence}: \\
\textit{ATC:} \callsign, number 2, following a (aircraft) on (distance) mile final, wind (degrees) at (speed), runway (rwy), cleared to land. \\
\textit{Pilot readback:} Number 2, following the (aircraft), runway (rwy), cleared to land, \callsign.

\item \textbf{Taxi-in}: \\
\textit{Pilot request:} Ground, \callsign, clear of runway (rwy), request taxi to parking. \\
\textit{ATC:} \callsign, taxi to (parking) via (twy’s), hold short/cross runway (rwy). \\
\textit{Pilot readback:} Taxi to (parking) via (twy’s), hold short/cross runway (rwy), \callsign.
\end{itemize}
