\section{IFR Phraseology}

\subsection*{IFR Clearance (RNAV SID)}

\textbf{Pilot request:} \control, \callsign, with information Bravo, request IFR to \destination~airport. \\

\textbf{ATC:} \callsign, cleared to \destination~airport, \SID~departure, then as filed. Climb via the sid, expect (cruise level) 10 minutes after departure. Departure frequency (freq), squawk \squawk. \\

\textbf{Pilot readback:} Cleared to \destination~airport via \SID~departure, then as filed. Climb via the SID, expect (cruise level) 10 minutes after departure, departure frequency (freq), squawk \squawk, \callsign.

\subsection*{IFR Clearance (Vectors)}

\textbf{Pilot request:} \control, \callsign, with information Bravo, request IFR to \destination~airport. \\

\textbf{ATC:} \callsign, cleared to \destination~airport, \SID~departure, radar vectors (fix), then as filed. Maintain (altitude), expect (cruise level) 10 minutes after departure. Departure frequency (freq), squawk \squawk. \\

\textbf{Pilot readback:} Cleared to \destination~airport via \SID~departure, radar vectors (fix), then as filed. Maintain (altitude), expect (cruise level) 10 minutes after departure, departure frequency (freq), squawk \squawk, \callsign.

\subsection*{Pushback (non-movement area)}

\textbf{Pilot request:} Ground, \callsign, stand/gate (gate), request push and start. \\

\textbf{ATC:} Readback correct, start and push at your discretion, expect runway (rwy) for departure, advise when ready to taxi. \\

\textbf{Pilot response:} Start and push at our discretion, expect runway (rwy), will advise ready to taxi, \callsign.

\subsection*{Pushback (movement area)}

\textbf{Pilot request:} Ground, \callsign, stand/gate (gate), request push and start. \\

\textbf{ATC:} Readback correct, start and push approved, face (direction), expect runway (rwy) for departure, advise when ready to taxi. \\

\textbf{Pilot response:} Start and push approved, face (direction), expect runway (rwy), will advise ready to taxi, \callsign.

\subsection*{Taxi}

\textbf{Pilot request:} Ground, \callsign, ready to taxi. \\

\textbf{ATC:} \callsign, runway (rwy), taxi via (twy’s), cross/hold short runway (rwy). \\

\textbf{Pilot readback:} Runway (rwy), taxi via (twy’s), cross/hold short runway (rwy), \callsign.

\subsection*{Line-up}

\textbf{Pilot request:} Tower, \callsign, ready for departure. \\

\textbf{ATC:} \callsign, runway (rwy), line up and wait. \\

\textbf{Pilot readback:} Runway (rwy), line up and wait, \callsign.

\subsection*{Line-up at intersection}

\textbf{Pilot request:} Tower, \callsign, ready for departure at (twy). \\

\textbf{ATC:} \callsign, runway (rwy) at (twy), line up and wait. \\

\textbf{Pilot readback:} Runway (rwy) at (twy), line up and wait, \callsign.

\subsection*{Takeoff (RNAV)}

\textbf{ATC:} \callsign, RNAV (initial fix), wind (degrees) at (speed). Runway (rwy), cleared for takeoff. \\

\textbf{Pilot readback:} RNAV (initial fix), runway (rwy), cleared for takeoff, \callsign.

\subsection*{Takeoff (Vectors)}

\textbf{ATC:} \callsign, fly heading (hdg), wind (degrees) at (speed). Runway (rwy), cleared for takeoff. \\

\textbf{Pilot readback:} Fly heading (hdg), runway (rwy), cleared for takeoff, \callsign.

\subsection*{Takeoff at intersection}

\textbf{ATC:} \callsign, fly heading (hdg), wind (degrees) at (speed). Runway (rwy) at (twy), cleared for takeoff. \\

\textbf{Pilot readback:} Fly heading (hdg), runway (rwy) at (twy), cleared for takeoff, \callsign.

\subsection*{Landing}

\textbf{ATC:} \callsign, wind (degrees) at (speed), runway (rwy), cleared to land. \\

\textbf{Pilot readback:} Runway (rwy), cleared to land, \callsign.

\subsection*{Landing w/ sequence}

\textbf{ATC:} \callsign, number 2, following a (aircraft) on (distance) mile final, wind (degrees) at (speed), runway (rwy), cleared to land. \\

\textbf{Pilot readback:} Number 2, following the (aircraft), runway (rwy), cleared to land, \callsign.

\subsection*{Taxi-in}

\textbf{Pilot request:} Ground, \callsign, clear of runway (rwy), request taxi to parking. \\

\textbf{ATC:} \callsign, taxi to (parking) via (twy’s), hold short/cross runway (rwy). \\

\textbf{Pilot readback:} Taxi to (parking) via (twy’s), hold short/cross runway (rwy), \callsign.

\section{IFR Example}

\subsection*{Departure: London Gatwick (EGKK)}

\subsubsection*{Delivery / Clearance}

\textbf{Pilot:} Gatwick Delivery, \callsign, Stand 552, Information Bravo, request IFR clearance to Amsterdam Schiphol.

\textbf{EGKK\_DEL:} \callsign, Gatwick Delivery, good morning. Cleared to Amsterdam Schiphol via the LAM 6M departure, flight level 120, squawk 4215.

\textbf{Pilot:} Cleared to Amsterdam Schiphol via LAM 6M departure, flight level 120, squawk 4215, \callsign.

\textbf{EGKK\_DEL:} \callsign, readback correct. Contact Gatwick Ground on 121.805 when ready for push and start.

\textbf{Pilot:} Ground on 121.805, \callsign.

\subsubsection*{Pushback \& Taxi}

\textbf{Pilot:} Gatwick Ground, \callsign, Stand 552, request push and start.

\textbf{EGKK\_GND:} \callsign, Gatwick Ground, push and start approved, facing tail south.

\textbf{Pilot:} Push and start approved, facing tail south, \callsign.

\textit{(After engine start and checklist completion)}

\textbf{Pilot:} \callsign, ready for taxi.

\textbf{EGKK\_GND:} \callsign, taxi to holding point Runway 26L via taxiways Juliet and Bravo, hold short of Runway 26L.

\textbf{Pilot:} Taxi to holding point Runway 26L via Juliet and Bravo, hold short Runway 26L, \callsign.

\textbf{EGKK\_GND:} \callsign, contact Gatwick Tower on 124.230 at the holding point.

\textbf{Pilot:} Tower on 124.230, \callsign.

\subsubsection*{Departure \& Takeoff}

\textbf{Pilot:} Gatwick Tower, \callsign, holding short Runway 26L, ready for departure.

\textbf{EGKK\_TWR:} \callsign, Gatwick Tower, line up and wait Runway 26L.

\textbf{Pilot:} Line up and wait Runway 26L, \callsign.

\textbf{EGKK\_TWR:} \callsign, wind 240 degrees 10 knots, Runway 26L, cleared for takeoff.

\textbf{Pilot:} Cleared for takeoff Runway 26L, \callsign.

\textit{(Airborne and climbing)}

\textbf{EGKK\_TWR:} \callsign, contact London Control on 126.950. Have a good flight.

\textbf{Pilot:} London Control on 126.950, goodbye, \callsign.

\subsection*{En Route: Radar Vectors \& Handoffs}

\subsubsection*{Area Control (Radar)}

\textbf{Pilot:} London Control, \callsign, passing 3,000 feet climbing to flight level 120 on the LAM 6M departure.

\textbf{LON\_CTR:} \callsign, London Control, radar contact. Climb to flight level 240.

\textbf{Pilot:} Climb to flight level 240, \callsign.

\textit{(Later during cruise)}

\textbf{LON\_CTR:} \callsign, direct to REDFA, expect descent in 10 miles.

\textbf{Pilot:} Direct REDFA, \callsign.

\textbf{LON\_CTR:} \callsign, contact Dutch Control on 125.750. Good day.

\textbf{Pilot:} Dutch Control on 125.750, \callsign.

\subsection*{Arrival: Amsterdam Schiphol (EHAM)}

\subsubsection*{Descent \& Approach}

\textbf{Pilot:} Dutch Control, \callsign, flight level 240, Information Delta.

\textbf{EHAM\_CTR:} \callsign, Dutch Control, descend flight level 100, QNH 1013, expect ILS approach Runway 18R via ARTIP 1A arrival.

\textbf{Pilot:} Descend flight level 100, QNH 1013, expecting ILS Runway 18R via ARTIP 1A arrival, \callsign.

\textbf{EHAM\_CTR:} \callsign, turn left heading 090, vectors for ILS Runway 18R, descend to 3,000 feet.

\textbf{Pilot:} Turn left heading 090, descend to 3,000 feet, \callsign.

\textbf{EHAM\_CTR:} \callsign, turn left heading 060, cleared ILS Runway 18R approach, report established.

\textbf{Pilot:} Turn left heading 060, cleared ILS Runway 18R approach, Wilco, \callsign.

\textit{(Once localizer is captured)}

\textbf{Pilot:} \callsign, established ILS Runway 18R.

\textbf{EHAM\_CTR:} \callsign, contact Schiphol Tower on 118.275.

\textbf{Pilot:} Tower on 118.275, \callsign.

\subsubsection*{Landing \& Arrival Taxi}

\textbf{Pilot:} Schiphol Tower, \callsign, established ILS Runway 18R.

\textbf{EHAM\_TWR:} \callsign, Schiphol Tower, wind 200 degrees 8 knots, Runway 18R, cleared to land.

\textbf{Pilot:} Cleared to land Runway 18R, \callsign.

\textit{(Touchdown and rollout)}

\textbf{EHAM\_TWR:} \callsign, vacate right via taxiway Victor, contact Schiphol Ground on 121.800.

\textbf{Pilot:} Vacate right Victor, Ground on 121.800, \callsign.

\textbf{Pilot:} Schiphol Ground, \callsign, vacated Runway 18R on Victor.

\textbf{EHAM\_GND:} \callsign, taxi to Stand E4 via taxiways Victor and Alpha.

\textbf{Pilot:} Taxi to Stand E4 via Victor and Alpha, \callsign.

\textit{(Arriving at gate)}

\textbf{EHAM\_GND:} \callsign, on stand E4, shutdown approved. Frequency change to Unicom 122.800. Have a nice day.

\textbf{Pilot:} Shutdown approved, Unicom 122.800, thanks for the service, goodbye!


\section{IVAO self-announcing}

\subsection*{Pre-Flight \& Engine Start}

\textit{Before starting engines, set COM1 to 122.800 MHz, check your surroundings visually and via TCAS, and announce your startup intentions.} \\

\textbf{Pilot announcement:} \departure~traffic, \callsign, Airbus A320, parked at stand 3, preparing for engine start, flight planned IFR to \destination, request weather check and starting up, \departure.

\subsection*{Taxi Out}

\textit{After starting engines and completing pre-taxi checklists:} \\

\textbf{Pilot announcement:} \departure~traffic, \callsign, taxiing from stand 3 to holding point runway 19 via taxiway Alpha, \departure. \\

\textit{Arriving at the holding point:} \\

\textbf{Pilot announcement:} \departure~traffic, \callsign, holding short of runway 19, performing final checks, will line up shortly, \departure.

\subsection*{Line-Up \& Takeoff}

\textit{Scan final approach and departure paths to ensure they are clear before entering the runway.} \\

\textbf{Pilot announcement:} \departure~traffic, \callsign, lining up and rolling runway 19, departing northbound, \departure.

\subsection*{Climb \& En-Route}

\textit{Leaving the initial altitude and climbing to Flight Level 140:} \\

\textbf{Pilot announcement:} All stations, \callsign, leaving altitude 3,000 feet climbing to Flight Level 140, direct to BPK, climbing out of \departure. \\

\textit{Reaching cruising level:} \\

\textbf{Pilot announcement:} All stations, \callsign, maintaining Flight Level 140, currently over BPK, proceeding direct to TNT.

\subsection*{Entering Airspace}

\textbf{Pilot announcement:} \callsign~entering EGTT FIR via LAM at FL350, transiting to north

\textbf{Pilot announcement:} \callsign~entering Paris TMA via MOT at FL120, descending to FL70 for arrival LFPG

\subsection*{Descent \& Approach Preparation}

\textit{Two to five minutes before commencing descent towards the destination:} \\

\textbf{Pilot announcement:} \destination~traffic, \callsign, Airbus A320, currently 40 miles south at Flight Level 140, commencing descent towards \destination~for an ILS approach runway 33, estimating arrival in 12 minutes. \\

\textit{Leaving cruising level:} \\

\textbf{Pilot announcement:} \destination~traffic, \callsign, leaving Flight Level 140 descending to Flight Level 80, inbound to \destination.

\subsection*{Approach, Landing \& Vacating}

\textit{Positioning for the approach:} \\

\textbf{Pilot announcement:} \destination~traffic, \callsign, descending through 4,000 feet, positioning for ILS runway 33 approach via transition point. \\

\textit{Established on the ILS / final approach:} \\

\textbf{Pilot announcement:} \destination~traffic, \callsign, established on ILS runway 33, 10 miles final, \destination. \\

\textit{On short final:} \\

\textbf{Pilot announcement:} \destination~traffic, \callsign, short final runway 33, full stop, \destination. \\

\textit{After landing, while vacating:} \\

\textbf{Pilot announcement:} \destination~traffic, \callsign, landed runway 33, vacating via taxiway Charlie.

\subsection*{Taxi to Gate \& Shutdown}

\textit{Once completely clear of the runway holding lines:} \\

\textbf{Pilot announcement:} \destination~traffic, \callsign, runway 33 fully vacated, taxiing to stand 42 via taxiway Alpha, \destination. \\

\textit{Arrived at the stand and shutting down:} \\

\textbf{Pilot announcement:} \destination~traffic, \callsign, parked at stand 42, engines shut down, flight complete. Thanks, and goodbye.
